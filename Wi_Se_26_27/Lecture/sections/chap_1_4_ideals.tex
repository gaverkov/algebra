A key notion in algebra of the rings. 

\begin{definition}[title = Definition 1.4.1]
    Let $(R, +, \cdot)$ be a commutative ring. Then a subset $I$ of $R$ is called an \textit{ideal} of $R$ if the following holds: 
    \begin{itemize}
      \item[(a)] $0 \in I$ 
      \item[(b)] For all $f, g \in I$, one has $f + g \in I$ 
      \item[(c)] For all $f \in R$ and $g \in I$, one has $f \cdot g \in I$   
    \end{itemize}
\end{definition}

The definition looks a little like the notion of a vector subspace.

You can view $I$ as a set of objects that are zero in some abstract sense. 

One can define ideals by system of generators. 

\begin{definition}[title = Definition 1.4.2]
    Let $R$ be a commutative ring and $f_1, \ldots, f_s$ finitely many elements of $R$. We define the ideal generated by $f_1, \ldots, f_s$ as follows: 
    \[\langle f_1, \ldots, f_s \rangle_{R} = \{f_1 \cdot g_1 + \ldots + f_s \cdot g_s \mid g_1, \ldots, g_s \in R\}\]  
    If the choice of $R$ is clear the subscript $R$ can be omitted.
\end{definition}

\begin{lemma}[title = Lemma 1.4.3]
    For $f_1, \ldots, f_s$ in a commutative ring $R$, the set $\langle f_1, \ldots, f_s \rangle_{R}$ is an ideal.

\begin{definition}[title = Definition 1.4.5]
    An ideal $I$ in a commutative ring $R$ is called \textit{finitely generated} if $I = \langle f_1, \ldots, f_s \rangle$ for some finitely many $f_1, \ldots, f_s \in R$. 
    In this case $\{f_1, \ldots, f_s\}$ is called a \textit{set of generators} of $I$.
\end{definition}

Now, we have an interplay: ideals vs. varieties. 
\[V(f_1, \ldots, f_s) \quad \quad \quad \langle f_1, \ldots, f_s \rangle\]
One can view the variety $V(f_1, \ldots, f_s)$ not as a function of $(f_1, \ldots, f_s)$ but as a function og the ideal $\langle f_1, \ldots, f_s, \rangle$.

\begin{definition}[title = Definition 1.4.7]
    For a set $X \subseteq k^n$ we introduce the \textit{ideal of $X$} by: 
    \[I(X) = \{f \in k[x_1, \dots, x_n] \mid f(a) = 0 \text{ for all } a \in X\}\]
\end{definition}

\begin{lemma}[title = Lemma 1.4.8]
    For $X \subseteq k^n$, the set $I(X)$ is in fact an ideal. 

    \paragraph{Proof:} Just spell out $I(X)$ and use the definition of an ideal. Try! 
\end{lemma}

\textbf{Short Recap} 

\begin{center}
    Variety $\quad \Leftrightarrow \quad$ Ideal
\end{center}

\[X \subseteq k^n \quad \mapsto \quad I(X) = \{f \in k[x_1, \dots, x_n]: f \text{ vanishes on } X\}\] 
\[V(f_1, \ldots, f_m) = \{a \in k^n: f_1(a) = \ldots = f_m(a) = 0\}\] 
\[\langle f_1, \ldots, f_m \rangle \quad \mapsto \quad V(f_1, \ldots, f_m) \subseteq k^n\quad \mapsto \quad \underbrace{I(V(f_1, \ldots, f_m))}_{\text{The set of polynomials, that vanish whenever the polynomials} f_1, \ldots, f_m \text{vanish.}}\]

One would frequently study a variety $V \subseteq k^n$ by studying the respective ideal $I(V)$. 

\begin{lemma}[title = Lemma 1.4.10]
    For $f_1, \ldots, f_s \in k[x_1, \dots, x_n]$, one has $\langle f_1, \ldots, f_s \rangle \subseteq I(V(f_1, \ldots, f_s))$ Moreover, there exist choices of $f_1, \ldots, f_s$ where whis inclusion is strict. 

    \paragraph{Proof:} \[g \in \langle f_1, \ldots, f_s \rangle \Leftrightarrow g = f_1 \cdot h_1 + \ldots + f_s \cdot h_s\]
    for some $h_1, \ldots, h_s \in k[x_1, \dots, x_n]$
\end{lemma}
\clearpage

\begin{lemma}[title = Lemma 1.4.10 continued]
    \begin{align*}
        g \in I(V(f_1, \ldots, f_s)) & \Leftrightarrow g \text{ vanishes on } V(f_1, \ldots, f_s) \\ 
                                     & \Leftrightarrow g(a) = 0 \text{ whenever } f_1(a)  = \ldots = f_s(a) = 0, \text{ for } a \in k^n. \\
    \end{align*}
    This explains the inclusion in the assertion. For strict inclusions, consider $k[x]$ with $V(x^2)$: 
    \begin{itemize}
      \item $\langle x^2 \rangle$ = set of all polynomials that do not have constant and linear terms. 
      \item $V(x^2) = \{0\} \quad \Rightarrow \quad I(V(x^))$ = set of polynomials that do not have the constant term.   
    \end{itemize}
    Consequently, $x \in I(V(x^2))$ but $x \in \langle x^2 \rangle$. 

    \begin{flushright}
      $\Box $
    \end{flushright}
\end{lemma} 

The so-called \textit{Nullstellensatz} clarifies the relation of $I(V(f_1, \ldots, f_s))$ to $\langle f_1, \ldots, f_s \rangle$ in the algebraic term in the 
case when the field $k$ is algebraically closed (for example $k = \mathbb{C}$).

\begin{proposition}[title = Proposition 1.4.11]
    Let $V,W \subseteq k^n$ be affine Varieties. Then the following hold: 
    \begin{itemize}
      \item[(i)] $V \subseteq W \Rightarrow I(V) \supseteq I(W)$. 
      \item[(ii)] $V = W \Leftrightarrow I(V) = I(W)$. 
    \end{itemize} 

    \paragraph{Proof:}  

    \begin{itemize}
      \item[(i)] follows direktly from our definition of $I(X)$ for a set $X \in k^n$. 
      \item[(ii)] $V = W \quad \Rightarrow I(V) = I(W)$ is trivial. \\
      
      Conversely, assume $I(V) = I(W)$. Let $V = V(f_1, \ldots, f_s)$ and $W = V(g_1, \ldots, g_t)$ mit $f_1, \ldots, f_s, g_1, \ldots, g_t \in k[x_1, \dots, x_n]$.

      Equality $I(V) = I(W)$ means that every polynomial vanishing on $V$ also vanishes on $W$ and vice versa. 
      \[f_1, \ldots, f_s \text{ vanishes on} V \Leftrightarrow g_1, \ldots, g_t \text{ vanishes on} W\] 
      So, $W \subseteq V(f_1, \ldots, f_s) = V$. Exchanging the roles of $V$ and $W$, we also have the other inlcusion $V \subseteq W$. 
      \[\Rightarrow V = W\]
    \end{itemize}
      \begin{flushright}
        $\Box $
      \end{flushright}
\end{proposition}
